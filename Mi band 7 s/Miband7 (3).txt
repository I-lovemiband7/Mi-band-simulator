  Barra espacio: flappy saltar<br>
  Q / E : bajar/subir volumen música<br>
  Scroll: mover apps largas

<script>
// =================== VARIABLES PRINCIPALES ===================
const pageContainer = document.getElementById("pageContainer");
const bg = document.getElementById("bg");
let currentApp = "watchface";
let bgOpacity = localStorage.getItem("bgOpacity") ? parseFloat(localStorage.getItem("bgOpacity")) : 0.7;
let bgImage = localStorage.getItem("bgImage") || "fondo1.jpg"; 

let audio = new Audio();
let playlist = ['beethoven.mp3','paganini.mp3'];
let currentSongIndex = -1;
let volume = 0.5; 

// =================== FONDO ===================
function updateBackground(){
  if(bgImage){
    bg.style.backgroundImage = `url(${bgImage})`;
    bg.style.opacity = bgOpacity;
  } else {
    bg.style.background = `linear-gradient(180deg, rgba(17,17,17,${bgOpacity}), rgba(51,51,51,${bgOpacity}))`;
  }
}
updateBackground();

// =================== CARGAR PÁGINA ===================
function load(html, appName, noAnim=false){
  pageContainer.innerHTML = `<div class="page ${noAnim?"nowatchface":""}" id="page">${html}</div>`;
  currentApp = appName;
}

// =================== SCROLL ===================
pageContainer.addEventListener("wheel", e=>{
  const page = document.getElementById("page");
  if(page){
    page.scrollTop += e.deltaY;
    e.preventDefault();
  }
},{passive:false});

// =================== WATCHFACE ===================
function watchface(){
  const now = new Date();
  load(`
    <div class="time">${now.getHours().toString().padStart(2,'0')}:${now.getMinutes().toString().padStart(2,'0')}</div>
    <div class="date">${now.toDateString()}</div>
    <div class="center"><div class="app" onclick="menu()">abrir apps</div></div>
  `,"watchface",true);
}
setInterval(()=>{ if(currentApp==="watchface") watchface(); },1000);

// =================== MENÚ ===================
function menu(){
  load(`
    <h2 class="center">apps</h2>
    <div class="app" onclick="pasos()">pasos</div>
    <div class="app" onclick="ritmo()">ritmo</div>
    <div class="app" onclick="musicApp()">música</div>
    <div class="app" onclick="social()">social</div>
    <div class="app" onclick="games()">juegos</div>
    <div class="app" onclick="settings()">configuración</div>
    <div class="back" onclick="watchface()">volver</div>
  `,"menu");
}

// =================== PASOS ===================
function pasos(){ 
  let steps = Math.floor(Math.random()*10000);
  let goal = 8000;
  let percent = Math.min(100, Math.round((steps/goal)*100));
  load(`
    <h2 class="center">pasos</h2>
    <p class="center">${steps} pasos</p>
    <div class="progress-bar"><div class="progress-bar-inner" style="width:${percent}%"></div></div>
    <p class="center">Meta: ${goal}</p>
    <div class="back" onclick="menu()">volver</div>
  `,"pasos"); 
}

// =================== RITMO ===================
function ritmo(){ 
  let bpm = 60 + Math.floor(Math.random()*60);
  let minBpm = 60, maxBpm = 120;
  let percent = Math.min(100, Math.round(((bpm-minBpm)/(maxBpm-minBpm))*100));
  load(`
    <h2 class="center">ritmo</h2>
    <p class="center">${bpm} bpm</p>
    <div class="progress-bar"><div class="progress-bar-inner" style="width:${percent}%"></div></div>
    <p class="center">Min: ${minBpm} Max: ${maxBpm}</p>
    <div class="back" onclick="menu()">volver</div>
  `,"ritmo"); 
}

// =================== MÚSICA ===================
function musicApp(){
  load(`
    <h2 class="center">música</h2>
    <div class="app" onclick="playSong(0)">novena sinfonía – beethoven</div>
    <div class="app" onclick="playSong(1)">la campanella – paganini</div>
    <div class="app" onclick="pauseSong()">pausar / reanudar</div>
    <p class="center" id="songName"></p>
    <p class="center" id="volumeDisplay">volumen: ${Math.round(volume*100)}%</p>
    <div class="back" onclick="stopMusic(); menu()">volver</div>
  `,"music");
  audio.volume = volume;
  currentSongIndex = -1;
}
function playSong(index){
  currentSongIndex = index;
  audio.src = playlist[index];
  audio.volume = volume;
  audio.play();
  document.getElementById('songName').textContent = index===0?"Novena Sinfonía – Beethoven":"La Campanella – Paganini";
  updateVolumeDisplay();
}
function pauseSong(){ if(audio){ audio.paused?audio.play():audio.pause(); }}
function stopMusic(){ if(audio){ audio.pause(); audio.currentTime=0; document.getElementById('songName')?.remove(); updateVolumeDisplay(); }}
function updateVolumeDisplay(){ const display = document.getElementById('volumeDisplay'); if(display) display.textContent = "Volumen: "+Math.round(volume*100)+"%"; }

// =================== JUEGOS ===================
function games(){
  load(`
    <h2 class="center">juegos</h2>
    <div class="app" onclick="flappy()">flappy</div>
    <div class="app" onclick="snake()">snake</div>
    <div class="back" onclick="menu()">volver</div>
  `,"games");
}

// =================== CONFIGURACIÓN ===================
function settings(){
  load(`
    <h2 class="center">configuración</h2>
    <div class="app" onclick="setBgOpacity(0.5)">transparencia baja</div>
    <div class="app" onclick="setBgOpacity(0.7)">transparencia media</div>
    <div class="app" onclick="setBgOpacity(0.9)">transparencia alta</div>
    <div class="app" onclick="setBgImage('fondo1.jpg')">fondo 1</div>
    <div class="app" onclick="setBgImage('fondo2.jpg')">fondo 2</div>
    <div class="app" onclick="setBgImage(null)">sin fondo</div>
    <div class="back" onclick="menu()">volver</div>
  `,"settings");
}

function setBgOpacity(value){
  bgOpacity = value;
  localStorage.setItem("bgOpacity", value);
  updateBackground();
  menu();
}

function setBgImage(file){
  bgImage = file;
  localStorage.setItem("bgImage", file);
  updateBackground();
  menu();
}

// =================== TECLADO ===================
document.addEventListener("keydown", e=>{
  if(e.key==='w' && currentApp==="watchface") menu(); 
  if(e.key==='s' && currentApp==="watchface") watchface();
  if(e.key==='x'){ if(currentApp!=="watchface" && currentApp!=="menu") menu(); }
  
  if(currentApp==="music"){
    if(e.key==='q'){ volume=Math.max(0,volume-0.1); if(audio) audio.volume=volume; updateVolumeDisplay();}
    if(e.key==='e'){ volume=Math.min(1,volume+0.1); if(audio) audio.volume=volume; updateVolumeDisplay();}
  }
});

// =================== INICIO ===================
watchface();
</script>
